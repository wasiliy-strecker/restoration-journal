cmake_minimum_required(VERSION 3.25)
project(RestaurierungsHeft VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(RESTAURIERUNGSHEFT_SANITIZERS "Enable address and undefined behavior checks" OFF)
include(CTest)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Sql)
if(BUILD_TESTING)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
endif()

add_library(project_options INTERFACE)
if(MSVC)
  target_compile_options(project_options INTERFACE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(RESTAURIERUNGSHEFT_SANITIZERS)
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(project_domain src/domain/project.cpp src/application/project_service.cpp)
target_include_directories(project_domain PUBLIC src)
target_link_libraries(project_domain PUBLIC Qt6::Core PRIVATE project_options)
add_library(project_photo src/application/photo_importer.cpp)
target_include_directories(project_photo PUBLIC src)
target_link_libraries(project_photo PUBLIC Qt6::Gui PRIVATE project_options)
add_library(project_storage src/persistence/sqlite_project_repository.cpp src/persistence/data_directory.cpp)
target_link_libraries(project_storage PUBLIC project_domain project_photo Qt6::Sql PRIVATE project_options)
add_library(project_pdf src/pdf/pdf_renderer.cpp src/pdf/pdf_export.cpp)
target_link_libraries(project_pdf PUBLIC project_domain project_photo Qt6::Gui PRIVATE project_options)
qt_add_resources(project_pdf export_fonts PREFIX "/" FILES assets/fonts/DejaVuSans.ttf)
add_library(project_ui src/ui/export_dialogs.cpp src/ui/checklist_editor.cpp src/ui/storage_controller.cpp src/ui/project_editor.cpp src/ui/main_window.cpp src/ui/ui_support.cpp src/ui/photo_view.cpp)
target_link_libraries(project_ui PUBLIC project_domain project_photo project_pdf Qt6::Widgets PRIVATE project_options)
qt_add_executable(restaurierungsheft WIN32 MACOSX_BUNDLE src/main.cpp)
target_link_libraries(restaurierungsheft PRIVATE project_storage project_ui project_options)
target_compile_definitions(restaurierungsheft PRIVATE APP_VERSION="${PROJECT_VERSION}")
qt_add_resources(restaurierungsheft notices PREFIX "/licenses" FILES THIRD_PARTY_NOTICES.md)
qt_add_resources(restaurierungsheft branding PREFIX "/" FILES assets/branding/icon-256.png)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  target_sources(restaurierungsheft PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
set_target_properties(restaurierungsheft PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER de.appfabrik.restaurierungsheft
  MACOSX_BUNDLE_BUNDLE_NAME RestaurierungsHeft
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in")

if(BUILD_TESTING)
  foreach(test_name domain repository dialogs photo status photo_dialogs checklist pdf pdf_dialogs)
    add_executable(${test_name}_tests tests/${test_name}_test.cpp)
    target_link_libraries(${test_name}_tests PRIVATE project_storage project_ui Qt6::Test project_options)
    add_test(NAME ${test_name} COMMAND ${test_name}_tests)
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
endif()

install(TARGETS restaurierungsheft BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION licenses)
install(DIRECTORY licenses/ DESTINATION licenses)
if(WIN32 OR APPLE)
  qt_generate_deploy_app_script(TARGET restaurierungsheft OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
